🔴 Part 1 — Critical (5건)

C-1. notifyActivity 커서 이동 기반 재정의

문제: 현재 _moveGateHeld == true만으로 활동 판정 → Top M 누르고 정지 상태에서도 idle 갱신, 반대로 Top M 놓고 흔들면 idle 미갱신

해결: 커서 실제 이동량 기반 idle 리셋

파일: E10_AirMouse_Task_0410.cpp::_sensorTask

Before:

```cpp
if (v_moveGateHeld) v_m->_power.notifyActivity(v_nowP);
```

After:

```cpp
// 커서 실제 이동 (x/y/wheel/pan 중 하나라도 비영)
const bool v_hasMotion =
    (v_xo != 0) || (v_yo != 0) ||
    (v_wheelY != 0) || (v_panX != 0);

if (v_hasMotion) {
    v_m->_power.notifyActivity(v_nowP);
}
```

주의: v_xo/v_yo는 파이프라인 최종단 계산 이후 → 위치 이동 필요 (또는 별도 _reqActivity 플래그로 deferred notify).

대안 (blocking 없이):

```cpp
// 프레임 push 직전에 활동 플래그만 세팅

    v_m->_powerNotifyPending = true;

// 다음 sensorTask 루프 상단에서 처리

    v_m->_powerNotifyPending = false;

}
```

파라미터: 없음
복잡도: 낮음
효과: 실사용 반영 (자세 유지 시 sleep 조기 진입, 손 떨림 시 sleep 방지)

C-4. Wake Backoff (Thrashing 방지)

문제: Wake 후 즉시 sleep 조건 재충족 시 초당 10회 이상 sleep-wake 반복 → 배터리 오히려 급소모

해결: Wake 후 최소 활성 시간 강제

파일: P20_Power_0410.cpp

After:

```cpp
// 클래스 멤버
uint32_t _lastWakeMs = 0;

bool CL_P20_Power::sleepNow(uint32_t p_nowMs) {
    if (_idleTimeoutMs == 0) return false;

    // [C-4] Wake 후 최소 활성 시간 강제
    if (_lastWakeMs != 0) {
        const uint32_t v_sinceWake = p_nowMs - _lastWakeMs;
        if (v_sinceWake < _wakeMinActiveMs) return false;
    }

    if ((p_nowMs - _lastActivityMs) < _idleTimeoutMs) return false;

    if (!_prepareMpuWom()) return false;
    _armExt1();

    esp_light_sleep_start();

    // wake
    _lastWakeMs = (uint32_t)millis();
    _lastActivityMs = _lastWakeMs;
    _idle = false;
    _ext1Armed = false;

    // [C-3] commTask 신호
    // ...

    return true;
}
```

파라미터:

이름 기본값 범위
wake_min_active_ms 500 100~2000

복잡도: 낮음
효과: Thrashing 완전 방지

```cpp
void CL_P20_Power::_prepareMpuWom() {
    // 1) PWR_MGMT_1: 내부 8MHz osc, sleep 해제
    mpuWr(G_MPU_PWR_MGMT_1, 0x00);
    delay(2);

    // 2) ACCEL_CFG: HPF 5Hz
    mpuWr(G_MPU_ACCEL_CFG, 0x01);

    // [C-5] Config 값 주입
    mpuWr(G_MPU_MOT_THR, _womThreshold);   // 기본 25
    mpuWr(G_MPU_MOT_DUR, _womDuration);    // 기본 4

    // 5) INT_PIN_CFG: LATCH | OPEN_DRAIN | ACTIVE_LOW
    mpuWr(G_MPU_INT_CFG, 0x80 | 0x40 | 0x20);

    // 6) INT_ENABLE: MOT_EN
    mpuWr(G_MPU_INT_EN, 0x40);

    // 7) PWR_MGMT_1: CYCLE=1, SLEEP=0
    mpuWr(G_MPU_PWR_MGMT_1, 0x20);

    // 8) PWR_MGMT_2: LP_WAKE_CTRL=5Hz, Gyro Standby
    mpuWr(G_MPU_PWR_MGMT_2, 0xC7);
    delay(5);

    (void)mpuRd(G_MPU_INT_STATUS);
    return true;
}
```

```cpp
const esp_sleep_wakeup_cause_t v_cause = esp_sleep_get_wakeup_cause();
const uint64_t v_status = esp_sleep_get_ext1_wakeup_status();

// Wake 원인 판정
enum EN_P20_WakeReason_t : uint8_t {
    EN_P20_WAKE_NONE       = 0,
    EN_P20_WAKE_MPU_MOTION = 1,
    EN_P20_WAKE_BUTTON     = 2,
    EN_P20_WAKE_OTHER      = 3,
};

EN_P20_WakeReason_t v_reason = EN_P20_WAKE_NONE;

if (v_cause == ESP_SLEEP_WAKEUP_EXT1) {
    if (v_status & (1ULL << 6)) {
        v_reason = EN_P20_WAKE_MPU_MOTION;
    } else {
        v_reason = EN_P20_WAKE_BUTTON;
    }
} else if (v_cause != ESP_SLEEP_WAKEUP_UNDEFINED) {
    v_reason = EN_P20_WAKE_OTHER;
}

// 원인별 처리
switch (v_reason) {
    case EN_P20_WAKE_MPU_MOTION:
        // 커서 이동 wake — LED 흰색 짧게
        _ledCb ? _ledCb(EN_L10_COLOR_WHITE, 100) : (void)0;
        break;
    case EN_P20_WAKE_BUTTON:
        // 버튼 wake — LED 조작 없음 (버튼 이벤트가 처리)
        break;
    case EN_P20_WAKE_OTHER:
        // 이상 wake — 로그
        D10_LOGW("[P20] unexpected wake cause: %d", (int)v_cause);
        break;
}

// 상태 저장 (진단)
_lastWakeReason = v_reason;
_wakeCount++;
```

```cpp
// L10_Led_0410.cpp
void CL_L10_Led::suspend(ST_LedSnapshot_t& p_out) {
    p_out.state        = _state;
    p_out.baseColor    = _baseColor;
    p_out.evtColor     = _evtColor;
    p_out.stateDurMs   = _stateDurMs;
    p_out.blinkPeriodMs= _blinkPeriodMs;
    p_out.stateStartMs = _stateStartMs;
    p_out.blinkOn      = _blinkOn;

    // 즉시 off
    _enterOff();
}

void CL_L10_Led::resume(const ST_LedSnapshot_t& p_in) {
    _baseColor    = p_in.baseColor;
    _evtColor     = p_in.evtColor;

    if (p_in.state == EN_L10_ST_OFF) {
        _enterOff();
        return;
    }

    // time-agnostic 복원: stateStartMs를 현재 시각 기준으로 시프트
    const uint32_t v_now = (uint32_t)millis();
    const uint32_t v_elapsed = (v_now >= p_in.stateStartMs)
        ? (v_now - p_in.stateStartMs) : 0;

    switch (p_in.state) {
        case EN_L10_ST_IDLE:
            _enterIdle(v_now);
            break;
        case EN_L10_ST_FLASH:
            if (v_elapsed < p_in.stateDurMs) {
                _state = EN_L10_ST_FLASH;
                _evtColor = p_in.evtColor;
                _stateStartMs = v_now - v_elapsed;
                _stateDurMs = p_in.stateDurMs;
                _apply(_evtColor, 255);
            } else {
                _enterIdle(v_now);
            }
            break;
        case EN_L10_ST_BLINK:
            // Blink는 무기한 또는 잔여 시간 있는 경우
            if (p_in.stateDurMs == 0 || v_elapsed < p_in.stateDurMs) {
                _state = EN_L10_ST_BLINK;
                _evtColor = p_in.evtColor;
                _blinkPeriodMs = p_in.blinkPeriodMs;
                _stateDurMs = p_in.stateDurMs;
                _stateStartMs = (p_in.stateDurMs == 0) ? v_now : (v_now - v_elapsed);
                _blinkOn = p_in.blinkOn;
            } else {
                _enterIdle(v_now);
            }
            break;
        default:
            _enterIdle(v_now);
            break;
    }
}
```

N-1. Deep-sleep 옵션

목적: 장기 미사용 시 초저전력 (Light-sleep ~0.8mA → Deep-sleep ~10µA)

파일: P20_Power_0410.h/.cpp

```cpp
bool CL_P20_Power::deepSleepNow(uint32_t p_nowMs) {
    if (_deepIdleTimeoutMs == 0) return false;

    // Deep-sleep 진입 조건
    if ((p_nowMs - _lastActivityMs) < _deepIdleTimeoutMs) return false;
    if (_hidConnected) return false;   // BLE 연결 중에는 금지
    if (_pairing) return false;

    D10_LOGI("[P20] entering deep-sleep");

    // 상태 저장 (재부팅 시 복원)
    // - active profile은 이미 LittleFS에 저장됨
    // - boot_state.pending = false로 (재부팅 성공 마킹)

    // LED off
    _ledCb ? _ledCb(...) : (void)0;

    // Wake 소스: 6버튼만 (MPU INT 불가)
    const uint64_t v_btnMask =
        (1ULL << 7)  | (1ULL << 12) | (1ULL << 13) |
        (1ULL << 14) | (1ULL << 15) | (1ULL << 16);
    esp_sleep_enable_ext1_wakeup(v_btnMask, ESP_EXT1_WAKEUP_ANY_LOW);

    esp_deep_sleep_start();
    return true;   // 도달 안 함 (deep-sleep 후 리셋)
}
```

파라미터:

이름 기본값 범위 비고
deep_idle_timeout_ms 600000 (10분) 0 또는 300000~3600000 0 = 비활성

Wake 후: 재부팅 (Boot 시퀀스 정상 진행)

복잡도: 높음 (Boot state 통합)
효과: 80배 전력 절감

N-4. 활동 프로파일 학습 (선택, Low priority)

목적: 시간대별 idle timeout 자동 조정

```cpp
uint16_t _activityHourHist[24];

// 매 활동 시
_activityHourHist[currentHour % 24]++;

// idle timeout 조정
uint32_t _getAdaptiveIdleTimeout(uint8_t p_mode) {
    const uint8_t v_hour = (millis() / 3600000) % 24;
    const uint16_t v_hist = _activityHourHist[v_hour];

    if (v_hist < 100) return _idleTimeoutMs[p_mode - 1] * 2;   // 활동 적음
    if (v_hist > 1000) return _idleTimeoutMs[p_mode - 1] / 2;  // 활동 많음
    return _idleTimeoutMs[p_mode - 1];
}
```

```cpp
    // stateStartMs 기준으로 경과 시간 계산
    const uint32_t v_elapsed = (v_now >= p_in.stateStartMs)
                             ? (v_now - p_in.stateStartMs) : 0;

    switch (p_in.state) {
        case EN_L10_ST_OFF:
            _enterOff();
            break;

        case EN_L10_ST_IDLE:
            _enterIdle(v_now);
            break;

                _state        = EN_L10_ST_FLASH;
                _evtColor     = p_in.evtColor;

                _stateDurMs   = p_in.stateDurMs;

            } else {
                _enterIdle(v_now);
            }
            break;

        case EN_L10_ST_BLINK:

                _evtColor      = p_in.evtColor;
                _blinkPeriodMs = p_in.blinkPeriodMs;
                _stateDurMs    = p_in.stateDurMs;

                _enterIdle(v_now);
            }

        case EN_L10_ST_FADEIN:
```

```cpp
bool CL_P20_Power::sleepNow(uint32_t p_nowMs) {
    // Config 검증
    if (_cfg.idle_timeout_ms[0] == 0 &&
        _cfg.idle_timeout_ms[1] == 0 &&
        _cfg.idle_timeout_ms[2] == 0) {
        return false;   // 모든 Mode 비활성
    }

    // [C-4] Wake Backoff — Wake 후 최소 활성 시간
    if (_lastWakeMs != 0) {
        const uint32_t v_sinceWake = p_nowMs - _lastWakeMs;
        if (v_sinceWake < _cfg.wake_min_active_ms) return false;
    }

    // Idle 경과 확인
    if ((p_nowMs - _lastActivityMs) < _cfg.idle_timeout_ms[0]) {
        // 최소값보다 작으면 아직 idle 아님 (Mode별 조회는 E10에서 사전 판정)
        return false;
    }

    if (!_prepareMpuWom()) return false;

    _armExt1();

    D10_LOGI("[P20] sleep (idle=%ums)", (unsigned)(p_nowMs - _lastActivityMs));

    esp_light_sleep_start();

    // ====================================================
    // [C-3, C-4] Wake 후 처리
    // ====================================================
    const uint32_t v_now = (uint32_t)millis();
    _lastWakeMs     = v_now;
    _lastActivityMs = v_now;
    _idle           = false;
    _ext1Armed      = false;

    // [I-1] Wake 원인 판정
    const esp_sleep_wakeup_cause_t v_cause = esp_sleep_get_wakeup_cause();
    const uint64_t v_status = esp_sleep_get_ext1_wakeup_status();

    if (v_cause == ESP_SLEEP_WAKEUP_EXT1) {
        if (v_status & (1ULL << 6)) {
            _lastWakeReason = EN_WAKE_MPU_MOTION;
        } else {
            _lastWakeReason = EN_WAKE_BUTTON;
        }
    } else {
        _lastWakeReason = EN_WAKE_OTHER;
    }
    _wakeCount++;

    _restoreMpuAfterWake();

    // [C-3] commTask 즉시 활성 신호
    if (_wakeCb) _wakeCb(_wakeCtx);

    D10_LOGI("[P20] wake: cause=%d status=0x%llx reason=%u cnt=%u",
             (int)v_cause, (unsigned long long)v_status,
             (unsigned)_lastWakeReason, (unsigned)_wakeCount);

    return true;
}
```

```cpp
bool CL_P20_Power::deepSleepNow(uint32_t p_nowMs,
                                bool p_hidConnected,
                                bool p_pairing) {
    // [N-1] Deep-sleep 진입 조건
    if (_cfg.deep_idle_timeout_ms == 0) return false;
    if (p_hidConnected) return false;      // BLE 연결 중 금지
    if (p_pairing) return false;            // Pairing 중 금지

    if ((p_nowMs - _lastActivityMs) < _cfg.deep_idle_timeout_ms) return false;

    D10_LOGI("[P20] entering DEEP-sleep (idle=%ums)",
             (unsigned)(p_nowMs - _lastActivityMs));

    // Wake 소스: 6버튼만 (MPU INT는 Deep-sleep에서 불가)
    const uint64_t v_btnMask =
        (1ULL << 7)  | (1ULL << 12) | (1ULL << 13) |
        (1ULL << 14) | (1ULL << 15) | (1ULL << 16);
    esp_sleep_enable_ext1_wakeup(v_btnMask, ESP_EXT1_WAKEUP_ANY_LOW);

    // Boot state 정상 마킹 (재부팅이 정상으로 인식되도록)
    // → E10에서 boot_state_0410.json의 pending=false 처리
    // (deep-sleep은 ESP 재부팅과 동일하므로 boot grace 통과 시 정상)

    esp_deep_sleep_start();
    return true;   // 도달 안 함
}
```

⚠️ 주의사항

1. _cfgProfile 참조 시점

_sensorTask에서 _cfgProfile.e10.power.* 참조 시 _cfgProfileValid == true 확인 필수:

```cpp
if (!v_m->_cfgProfileValid) { /* skip power logic */ }
```

2. Deep-sleep 부팅 후 상태

Deep-sleep은 ESP32 재부팅과 동일 → boot grace 재판정:

· boot_state_0410.json의 pending=false 유지 (Wake 성공 마킹)
· tickConfigSave()에서 자동 저장됨

3. LED suspend는 반드시 sleep 직전

Sleep 조건 판정 후 LED suspend → sleepNow → LED resume 순서.

4. commTask ulTaskNotifyTake

기존 vTaskDelay(7) 대체. Wake 신호 없을 때도 7ms 대기 (동일).

5. _wakeCb 등록 시점

_sensorTask 진입 시 1회 등록. begin()에서 하면 _thComm 미생성 시점.

6. 검증 우선순위

· 필수: C-4 (Thrashing 방지), I-8 (매크로)
· 권장: C-1, C-5, I-7
· 선택: N-1 (Deep-sleep, 실측 후 조정)
